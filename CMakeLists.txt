cmake_minimum_required(VERSION 3.16)
project(VTLC VERSION 0.1 LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VTLC_ZSTD "Enable Zstandard residual coding and baselines" ON)
if(VTLC_ZSTD)
  find_library(ZSTD_LIBRARY zstd REQUIRED)
endif()
add_library(vtlc SHARED src/codec.cpp src/streaming.cpp)
target_link_libraries(vtlc PRIVATE Threads::Threads)
target_include_directories(vtlc PUBLIC include)
if(VTLC_ZSTD)
  target_link_libraries(vtlc PRIVATE ${ZSTD_LIBRARY})
  target_compile_definitions(vtlc PRIVATE VTLC_HAVE_ZSTD=1)
endif()
target_compile_options(vtlc PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -frounding-math -ffp-contract=off)
option(VTLC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(VTLC_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(vtlc PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vtlc PRIVATE -fsanitize=address,undefined)
endif()
add_executable(vtlc_test tests/test.cpp)
target_link_libraries(vtlc_test PRIVATE vtlc)
target_compile_options(vtlc_test PRIVATE -UNDEBUG)
if(VTLC_ZSTD)
  add_executable(vtlc_bench benchmarks/benchmark.cpp)
  target_link_libraries(vtlc_bench PRIVATE vtlc ${ZSTD_LIBRARY})
endif()
enable_testing()
add_test(NAME roundtrip COMMAND vtlc_test)
install(TARGETS vtlc LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
add_executable(vtlc_edge_test tests/edge.cpp)
target_link_libraries(vtlc_edge_test PRIVATE vtlc)
target_compile_options(vtlc_edge_test PRIVATE -UNDEBUG)
add_test(NAME edge_cases COMMAND vtlc_edge_test)
add_executable(vtlc_stream_test tests/streaming.cpp)
target_link_libraries(vtlc_stream_test PRIVATE vtlc)
target_compile_options(vtlc_stream_test PRIVATE -UNDEBUG)
add_test(NAME streaming COMMAND vtlc_stream_test)
add_executable(vtlc_fma_test tests/fma.cpp)
target_link_libraries(vtlc_fma_test PRIVATE vtlc)
target_compile_options(vtlc_fma_test PRIVATE -UNDEBUG)
add_test(NAME fused_prediction COMMAND vtlc_fma_test)
add_executable(vtlc_bitshuffle_test tests/bitshuffle.cpp)
target_link_libraries(vtlc_bitshuffle_test PRIVATE vtlc)
target_compile_options(vtlc_bitshuffle_test PRIVATE -UNDEBUG)
add_test(NAME bitshuffle COMMAND vtlc_bitshuffle_test)
if(VTLC_ZSTD)
  add_executable(vtlc_codecs_test tests/codecs.cpp)
  target_link_libraries(vtlc_codecs_test PRIVATE vtlc ${ZSTD_LIBRARY})
  target_compile_options(vtlc_codecs_test PRIVATE -UNDEBUG)
  add_test(NAME residual_codecs COMMAND vtlc_codecs_test)
endif()
add_executable(vtlc_kernels benchmarks/kernels.cpp)
target_link_libraries(vtlc_kernels PRIVATE vtlc)
add_executable(vtlc_ratio_test tests/ratio.cpp)
target_link_libraries(vtlc_ratio_test PRIVATE vtlc)
target_compile_options(vtlc_ratio_test PRIVATE -UNDEBUG)
add_test(NAME ratio_roundtrip COMMAND vtlc_ratio_test)
option(VTLC_FUZZ "Build libFuzzer decoder target with Clang" OFF)
if(VTLC_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "VTLC_FUZZ requires Clang")
  endif()
  target_compile_options(vtlc PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(vtlc_fuzz fuzz/decode.cpp)
  target_link_libraries(vtlc_fuzz PRIVATE vtlc)
  target_compile_options(vtlc_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(vtlc_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
add_executable(vtlc_c_test tests/c_api.c)
target_link_libraries(vtlc_c_test PRIVATE vtlc)
target_compile_options(vtlc_c_test PRIVATE -UNDEBUG)
add_test(NAME c_interface COMMAND vtlc_c_test)
add_executable(vtlc_packed_test tests/packed.cpp)
target_link_libraries(vtlc_packed_test PRIVATE vtlc)
target_compile_options(vtlc_packed_test PRIVATE -UNDEBUG)
add_test(NAME vector_packing COMMAND vtlc_packed_test)
add_executable(vtlc_alignment_test tests/alignment.cpp)
target_link_libraries(vtlc_alignment_test PRIVATE vtlc)
target_compile_options(vtlc_alignment_test PRIVATE -UNDEBUG)
add_test(NAME alignment COMMAND vtlc_alignment_test)
